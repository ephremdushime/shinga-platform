Next.js Auth Implementation
Drop these files into your Next.js App Router project. Uses Supabase Auth with phone OTP via Africa's Talking.
lib/shinga/auth.ts
// ═══════════════════════════════════════════════════════
// SHINGA — Auth Library
// Phone OTP via Supabase Auth + Africa's Talking
// ═══════════════════════════════════════════════════════
import { createServerClient, createBrowserClient } from './supabase'
import { cookies }                                   from 'next/headers'
import { redirect }                                  from 'next/navigation'

// ── GET CURRENT USER (server) ─────────────────────────
export async function getUser() {
  const supabase = await createServerClient()
  const { data: { user } } = await supabase.auth.getUser()
  return user
}

// ── GET USER + PROFILE ────────────────────────────────
export async function getProfile() {
  const supabase = await createServerClient()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) return null
  const { data } = await supabase
    .from('profiles')
    .select('*, district:districts(name, province)')
    .eq('id', user.id)
    .single()
  return data
}

// ── REQUIRE AUTH (server component guard) ─────────────
export async function requireAuth(redirectTo = '/login') {
  const user = await getUser()
  if (!user) redirect(redirectTo)
  return user
}

// ── REQUIRE ROLE ──────────────────────────────────────
export async function requireRole(role: 'worker' | 'employer' | 'admin') {
  const profile = await getProfile()
  if (!profile || profile.role !== role) redirect(`/dashboard`)
  return profile
}

// ── SEND OTP ──────────────────────────────────────────
// Supabase handles OTP delivery — configure AT in dashboard
export async function sendOtp(phone: string) {
  const supabase = createBrowserClient()
  // Supabase formats: +250XXXXXXXXX
  const formatted = phone.startsWith('+') ? phone : `+250${phone.replace(/\D/g,'')}`
  const { error } = await supabase.auth.signInWithOtp({ phone: formatted })
  return { error }
}

// ── VERIFY OTP ────────────────────────────────────────
export async function verifyOtp(phone: string, token: string) {
  const supabase = createBrowserClient()
  const formatted = phone.startsWith('+') ? phone : `+250${phone.replace(/\D/g,'')}`
  const { data, error } = await supabase.auth.verifyOtp({
    phone: formatted, token, type: 'sms'
  })
  return { user: data.user, error }
}

// ── SIGN OUT ──────────────────────────────────────────
export async function signOut() {
  const supabase = createBrowserClient()
  await supabase.auth.signOut()
  redirect('/login')
}

// ── CREATE PROFILE AFTER SIGNUP ───────────────────────
export async function createProfile(data: {
  userId:    string
  fullName:  string
  phone:     string
  role:      'worker' | 'employer'
  districtId?: number
}) {
  const supabase = await createServerClient()
  const { error } = await supabase.from('profiles').upsert({
    id:             data.userId,
    full_name:      data.fullName,
    phone:          data.phone,
    phone_verified: true,
    role:           data.role,
    district_id:    data.districtId,
  })
  return { error }
}

// ── CREATE WORKER RECORD ──────────────────────────────
export async function createWorkerProfile(data: {
  profileId:   string
  fullName:    string
  phone:       string
  category:    string
  skillTitle:  string
  skillTags:   string[]
  bio:         string
  rateAmount:  number
  rateUnit:    string
  rateNeg:     boolean
  availability:string
  districtId:  number
  sector:      string
  canTravel:   boolean
  yearsExp:    number
}) {
  const supabase = await createServerClient()
  const { error } = await supabase.from('workers').insert({
    profile_id:    data.profileId,
    full_name:     data.fullName,
    phone:         data.phone,
    category:      data.category,
    skill_title:   data.skillTitle,
    skill_tags:    data.skillTags,
    bio:           data.bio,
    rate_amount:   data.rateAmount,
    rate_unit:     data.rateUnit,
    rate_negotiable: data.rateNeg,
    availability:  data.availability,
    district_id:   data.districtId,
    sector:        data.sector,
    can_travel:    data.canTravel,
    years_exp:     data.yearsExp,
    status:        'pending_review',
  })
  return { error }
}
app/api/auth/send-otp/route.ts
import { NextRequest, NextResponse } from 'next/server'
import { sendOtp } from '@/lib/shinga/auth'

export async function POST(req: NextRequest) {
  const { phone } = await req.json()
  if (!phone) return NextResponse.json({ error: 'Phone required' }, { status: 400 })

  const { error } = await sendOtp(phone)
  if (error) return NextResponse.json({ error: error.message }, { status: 400 })

  return NextResponse.json({ sent: true })
}
app/api/auth/verify-otp/route.ts
import { NextRequest, NextResponse } from 'next/server'
import { verifyOtp, createProfile } from '@/lib/shinga/auth'

export async function POST(req: NextRequest) {
  const { phone, token, fullName, role, districtId } = await req.json()

  const { user, error } = await verifyOtp(phone, token)
  if (error || !user) return NextResponse.json({ error: error?.message || 'Verification failed' }, { status: 400 })

  // Create or update profile
  await createProfile({ userId: user.id, fullName, phone, role, districtId })

  // New users → onboarding. Returning users → dashboard.
  const isNew = (Date.now() - new Date(user.created_at).getTime()) < 10_000
  const redirectTo = isNew
    ? role === 'worker' ? '/onboarding/worker' : '/onboarding/employer'
    : role === 'worker' ? '/dashboard/worker'  : '/dashboard/employer'

  return NextResponse.json({ redirectTo })
}
app/api/auth/complete-worker/route.ts
import { NextRequest, NextResponse } from 'next/server'
import { requireAuth, createWorkerProfile } from '@/lib/shinga/auth'
import { notify } from '@/lib/shinga/sms'

export async function POST(req: NextRequest) {
  const user = await requireAuth()
  const body = await req.json()

  const { error } = await createWorkerProfile({ profileId: user.id, ...body })
  if (error) return NextResponse.json({ error: error.message }, { status: 400 })

  // Welcome SMS
  await notify({
    to:    body.phone,
    event: 'application_sent',  // reuse as welcome
    data:  { job: 'SHINGA' },
    userId: user.id,
  })

  return NextResponse.json({ redirectTo: '/dashboard/worker' })
}
middleware.ts — route protection
import { NextRequest, NextResponse } from 'next/server'
import { createServerClient }         from '@/lib/shinga/supabase'

const PUBLIC_PATHS = ['/', '/login', '/register', '/workers', '/jobs']
const WORKER_PATHS = ['/dashboard/worker']
const EMPLOYER_PATHS = ['/dashboard/employer']
const ADMIN_PATHS = ['/admin']

export async function middleware(req: NextRequest) {
  const supabase  = await createServerClient()
  const { data: { user } } = await supabase.auth.getUser()
  const path      = req.nextUrl.pathname

  // Public paths — always allowed
  if (PUBLIC_PATHS.some(p => path.startsWith(p))) return NextResponse.next()

  // Protected — must be logged in
  if (!user) return NextResponse.redirect(new URL('/login', req.url))

  // Fetch role
  const { data: profile } = await supabase
    .from('profiles').select('role').eq('id', user.id).single()
  const role = profile?.role

  // Role-gated paths
  if (ADMIN_PATHS.some(p => path.startsWith(p)) && role !== 'admin')
    return NextResponse.redirect(new URL('/', req.url))
  if (WORKER_PATHS.some(p => path.startsWith(p)) && role !== 'worker')
    return NextResponse.redirect(new URL('/dashboard/employer', req.url))
  if (EMPLOYER_PATHS.some(p => path.startsWith(p)) && role !== 'employer')
    return NextResponse.redirect(new URL('/dashboard/worker', req.url))

  return NextResponse.next()
}

export const config = {
  matcher: ['/((?!_next/static|_next/image|favicon.ico|api).*)' ]
