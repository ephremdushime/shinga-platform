Dynamic Data Layer

Replaces all hardcoded arrays in the HTML prototypes with live Supabase data. Covers jobs, workers, search, the worker profile dynamic route, and auth-gated dashboard data for both employers and workers.

Next.js Page / Component
React Server Component or Client with SWR
→
API Route / Server Action
app/api/shinga/… + lib/shinga/…
→
Supabase (Postgres + RLS)
All 12 tables, search functions, triggers
✅ What this replaces
Every const JOBS = [...] and const WORKERS = [...] hardcoded array in shinga.html, shinga-employer.html, and shinga-worker-profile.html gets replaced by a real Supabase query through these files. The UI components stay exactly the same — only the data source changes.

Jobs API

lib/shinga/jobs.ts
// ═══════════════════════════════════════════════════════
// SHINGA — Jobs Data Layer
// ═══════════════════════════════════════════════════════
import { createServerClient } from './supabase'

export interface Job {
  id:                string
  employer_id:       string
  title:             string
  company_name:      string | null
  description:       string
  category:          string
  job_type:          string
  district_id:       number
  district:          { name: string; province: string }
  pay_amount:        number | null
  pay_unit:          string | null
  slots:             number
  featured:          boolean
  urgent:            boolean
  status:            string
  applications_count:number
  views_count:       number
  expires_at:        string
  created_at:        string
}

export interface JobFilters {
  category?:  string
  job_type?:  string
  district?:  string
  keyword?:   string
  featured?:  boolean
  page?:      number
  limit?:     number
}

// ── LIST JOBS ─────────────────────────────────────────
export async function getJobs(filters: JobFilters = {}) {
  const supabase = await createServerClient()
  const { page = 1, limit = 20 } = filters
  const from = (page - 1) * limit
  const to   = from + limit - 1

  let query = supabase
    .from('jobs')
    .select(`
      *,
      district:districts(name, province),
      employer:profiles(full_name)
    `, { count: 'exact' })
    .eq('status', 'active')
    .order('featured', { ascending: false })
    .order('created_at', { ascending: false })
    .range(from, to)

  if (filters.category) query = query.eq('category', filters.category)
  if (filters.job_type) query = query.eq('job_type', filters.job_type)
  if (filters.featured) query = query.eq('featured', true)
  if (filters.district) {
    // join filter on district name
    const { data: dist } = await supabase
      .from('districts').select('id').ilike('name', `%${filters.district}%`).single()
    if (dist) query = query.eq('district_id', dist.id)
  }
  if (filters.keyword) {
    query = query.or(
      `title.ilike.%${filters.keyword}%,description.ilike.%${filters.keyword}%`
    )
  }

  const { data, count, error } = await query
  return { jobs: data ?? [], total: count ?? 0, error }
}

// ── SINGLE JOB ────────────────────────────────────────
export async function getJob(id: string) {
  const supabase = await createServerClient()
  const { data, error } = await supabase
    .from('jobs')
    .select(`*, district:districts(name, province), employer:profiles(full_name, phone)`)
    .eq('id', id)
    .single()

  // Increment view count (fire and forget)
  if (data) {
    supabase.from('jobs')
      .update({ views_count: (data.views_count ?? 0) + 1 })
      .eq('id', id).then(() => {})
  }
  return { job: data, error }
}

// ── EMPLOYER'S JOBS ───────────────────────────────────
export async function getEmployerJobs(employerId: string) {
  const supabase = await createServerClient()
  const { data, error } = await supabase
    .from('jobs')
    .select(`*, district:districts(name), applications:applications(count)`)
    .eq('employer_id', employerId)
    .order('created_at', { ascending: false })
  return { jobs: data ?? [], error }
}

// ── CREATE JOB (after payment confirmed) ──────────────
export async function createJob(employerId: string, data: Partial<Job>) {
  const supabase = await createServerClient()
  const { data: job, error } = await supabase
    .from('jobs')
    .insert({ ...data, employer_id: employerId, status: 'active' })
    .select()
    .single()
  return { job, error }
}

// ── UPDATE JOB STATUS ─────────────────────────────────
export async function updateJobStatus(
  jobId: string, employerId: string,
  status: 'active' | 'paused' | 'filled'
) {
  const supabase = await createServerClient()
  const { error } = await supabase
    .from('jobs')
    .update({ status })
    .eq('id', jobId)
    .eq('employer_id', employerId)  // RLS: only own jobs
  return { error }
}
app/api/shinga/jobs/route.ts
import { NextRequest, NextResponse } from 'next/server'
import { getJobs, createJob } from '@/lib/shinga/jobs'
import { getUser }            from '@/lib/shinga/auth'

// GET /api/shinga/jobs?category=domestic&job_type=daily&keyword=cleaner&page=1
export async function GET(req: NextRequest) {
  const p        = req.nextUrl.searchParams
  const { jobs, total, error } = await getJobs({
    category:  p.get('category')  || undefined,
    job_type:  p.get('job_type')  || undefined,
    district:  p.get('district')  || undefined,
    keyword:   p.get('keyword')   || undefined,
    featured:  p.get('featured') === 'true',
    page:      parseInt(p.get('page') || '1'),
    limit:     parseInt(p.get('limit') || '20'),
  })
  if (error) return NextResponse.json({ error: error.message }, { status: 400 })
  return NextResponse.json({ jobs, total }, {
    headers: { 'Cache-Control': 'public, s-maxage=60, stale-while-revalidate=300' }
  })
}

// POST /api/shinga/jobs — create a job (after payment)
export async function POST(req: NextRequest) {
  const user = await getUser()
  if (!user) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
  const body = await req.json()
  const { job, error } = await createJob(user.id, body)
  if (error) return NextResponse.json({ error: error.message }, { status: 400 })
  return NextResponse.json({ job }, { status: 201 })
}

// ── SINGLE WORKER (public profile) ────────────────────
export async function getWorker(id: string) {
  const supabase = await createServerClient()
  const { data: worker, error } = await supabase
    .from('workers')
    .select(`
      *,
      district:districts(name, province),
      reviews:reviews(
        id, rating, comment, created_at,
        reviewer:profiles(full_name)
      )

    .eq('id', id)
    .eq('status', 'active')
    .single()

  // Increment view count
  if (worker) {
    supabase.from('workers')
      .update({ views_count: (worker.views_count ?? 0) + 1 })
      .eq('id', id).then(() => {})
  }
  return { worker, error }
}

// ── WORKER BY PROFILE ID (for dashboard) ──────────────
export async function getWorkerByProfileId(profileId: string) {

  const { data, error } = await supabase
    .from('workers')
    .select(`*, district:districts(name)`)
    .eq('profile_id', profileId)
    .single()
  return { worker: data, error }
}

// ── WORKER'S APPLICATIONS ─────────────────────────────
export async function getWorkerApplications(workerId: string) {
  const supabase = await createServerClient()
  const { data, error } = await supabase
    .from('applications')
    .select(`
      *,
      job:jobs(title, job_type, pay_amount, pay_unit, status,
        district:districts(name),
        employer:profiles(full_name)
      )
    `)
    .eq('worker_id', workerId)
    .order('created_at', { ascending: false })
  return { applications: data ?? [], error }
}

// ── SUBMIT APPLICATION (after payment) ────────────────
export async function submitApplication(data: {
  jobId:    string
  workerId: string
  note?:    string
}) {
  const supabase = await createServerClient()
  const { data: app, error } = await supabase
    .from('applications')
    .insert({
      job_id:    data.jobId,
      worker_id: data.workerId,
      note:      data.note,
      status:    'pending',
    })
    .select()
    .single()
  return { application: app, error }
}

// ── SIMILAR WORKERS (same category, exclude self) ─────
export async function getSimilarWorkers(workerId: string, category: string, districtId: number) {
  const supabase = await createServerClient()
  const { data } = await supabase
    .from('workers')
    .select(`id, full_name, skill_title, rate_amount, rate_unit, verified, rating_avg, district:districts(name)`)
    .eq('category', category)
    .eq('status', 'active')
    .neq('id', workerId)
    .order('verified', { ascending: false })
    .order('rating_avg', { ascending: false })
    .limit(4)
  return data ?? []
}

Search API

app/api/shinga/search/route.ts
// Unified search — hits Postgres full-text search functions from schema
import { NextRequest, NextResponse } from 'next/server'
import { createServerClient }        from '@/lib/shinga/supabase'

export async function GET(req: NextRequest) {
  const p       = req.nextUrl.searchParams
  const q       = p.get('q') || ''
  const type    = p.get('type') || 'jobs'   // 'jobs' | 'workers'
  const districtId = p.get('district_id') ? parseInt(p.get('district_id')!) : null
  const category   = p.get('category') || null

  const supabase = await createServerClient()

  if (type === 'jobs') {
    // Uses search_jobs() Postgres function from shinga-schema.sql
    const { data, error } = await supabase
      .rpc('search_jobs', {
        search_query: q,
        p_category:   category,
        p_district_id: districtId,
        p_limit:      20,
        p_offset:     0,
      })
    if (error) return NextResponse.json({ error: error.message }, { status: 400 })
    return NextResponse.json({ results: data })
  }

  if (type === 'workers') {
    // Uses search_workers() Postgres function from shinga-schema.sql
    const { data, error } = await supabase
      .rpc('search_workers', {
        search_query:  q,
        p_category:    category,
        p_district_id: districtId,
        p_limit:       20,
        p_offset:      0,
      })
    if (error) return NextResponse.json({ error: error.message }, { status: 400 })
    return NextResponse.json({ results: data })
  }

  return NextResponse.json({ error: 'Invalid type' }, { status: 400 })
}

SWR Client Hooks

hooks/shinga/useJobs.ts
'use client'
import useSWR from 'swr'

const fetcher = (url: string) => fetch(url).then(r => r.json())

export function useJobs(filters: Record<string, string> = {}) {
  const params = new URLSearchParams(
    Object.fromEntries(Object.entries(filters).filter(([,v]) => v))
  )
  const { data, error, isLoading, mutate } = useSWR(
    `/api/shinga/jobs?${params}`,
    fetcher,
    { revalidateOnFocus: false }
  )
  return {
    jobs:      (data?.jobs ?? []) as Job[],
    total:     data?.total ?? 0,
    isLoading,
    error,
    mutate,
  }
}

export function useJob(id: string) {
  const { data, error, isLoading } = useSWR(
    id ? `/api/shinga/jobs/${id}` : null,
    fetcher
  )
  return { job: data?.job ?? null, isLoading, error }
}

  )

export function useSearch(query: string, type: 'jobs' | 'workers', filters = {}) {
  const params = new URLSearchParams({ q: query, type, ...filters })
  const { data, error, isLoading } = useSWR(
    query.length > 1 ? `/api/shinga/search?${params}` : null,
    fetcher,
    { dedupingInterval: 300 }
  )
  return { results: data?.results ?? [], isLoading, error }
}

interface Props { params: { id: string } }

// Server Component — fetches data, passes to client
export default async function WorkerProfilePage({ params }: Props) {
  const { worker, error } = await getWorker(params.id)
  if (error || !worker) notFound()

  const [similar, viewer] = await Promise.all([
    getSimilarWorkers(worker.id, worker.category, worker.district_id),
    getProfile(),
  ])

    <WorkerProfileClient
      worker={worker}
      similar={similar}
      viewerRole={viewer?.role ?? null}
      viewerId={viewer?.id ?? null}
    />
  )
}

// Static params for ISR — pre-render top 100 workers
export async function generateStaticParams() {
  const { createServerClient } = await import('@/lib/shinga/supabase')
  const supabase = await createServerClient()
  const { data } = await supabase
    .from('workers')
    .select('id')
    .eq('status', 'active')
    .order('rating_avg', { ascending: false })
    .limit(100)
  return (data ?? []).map(w => ({ id: w.id }))
}

Worker Profile Client Component

app/workers/[id]/WorkerProfileClient.tsx
'use client'
import { useState }       from 'react'
import { usePayment }    from '@/hooks/shinga/usePayment'
import { Worker }        from '@/lib/shinga/workers'

interface Props {
  worker:     Worker
  similar:    Partial<Worker>[]
  viewerRole: 'employer' | 'worker' | 'admin' | null
  viewerId:   string | null
}

export function WorkerProfileClient({ worker, similar, viewerRole, viewerId }: Props) {
  const [revealed,   setRevealed]   = useState(false)
  const [phone,      setPhone]       = useState('')
  const [activeTab,  setActiveTab]   = useState('about')
  const [showContact,setShowContact] = useState(false)

  const { pay, status: payStatus } = usePayment({
    onSuccess: async (referenceId) => {
      // Fetch revealed phone from API (phone only returned after payment verified)
      const res  = await fetch(`/api/shinga/workers/${worker.id}/contact?ref=${referenceId}`)
      const data = await res.json()
      if (data.phone) { setPhone(data.phone); setRevealed(true) }
    },
    onFailure: (err) => console.error(err),
  })

  const handleContact = (myPhone: string) => {
    if (!viewerId) { window.location.href = '/login'; return }
    pay({
      action:   'contact_worker',
      phone:    myPhone,
      metadata: { worker_id: worker.id, worker_name: worker.full_name },
    })
  }

  const fmt = (n: number) => `RWF ${n.toLocaleString()}`

  return (
    <div className="worker-profile">

      {/* ── HERO ── */}
      <div className="profile-hero">
        <div className="ph-avi" style={{ background: `linear-gradient(135deg,${avatarColor(worker.full_name)})` }}>
          {initials(worker.full_name)}
          {worker.verified && <div className="ph-verif-ring">✓</div>}
        </div>
        <div className="ph-info">
          <h1>{worker.full_name}</h1>
          <div className="ph-role">{worker.skill_title}</div>
          <div className="ph-meta">
            <span>📍 {worker.district?.name}</span>
            <span>★ {worker.rating_avg} ({worker.rating_count} reviews)</span>
            <span>{worker.jobs_completed} jobs done</span>
          </div>
        </div>
        <div className="ph-actions">
          <div className="ph-rate">
            {worker.rate_amount ? `${fmt(worker.rate_amount)} / ${worker.rate_unit}` : 'Rate negotiable'}
          </div>
          {viewerRole === 'employer' || viewerRole === null ? (
            revealed ? (
              <a href={`tel:${phone}`} className="btn btn-green">📞 {phone}</a>
            ) : payStatus === 'pending' ? (
              <div className="payment-pending">
                <div className="spinner" />
                <p>Approve MoMo payment…</p>
              </div>
            ) : (
              <button className="btn btn-green" onClick={() => setShowContact(true)}>
                Contact — RWF 1,500
              </button>
            )
          ) : null}
        </div>
      </div>

      {/* ── TABS ── */}
      <div className="profile-tabs">
        {['about', 'reviews', 'availability'].map(tab => (
          <button key={tab} className={activeTab === tab ? 'active' : ''}
            onClick={() => setActiveTab(tab)}>{tab}</button>
        ))}
      </div>

      {/* ── TAB CONTENT ── */}
      {activeTab === 'about' && (
        <div>
          <p>{worker.bio}</p>
          <div className="skills-list">
            {worker.skill_tags?.map(tag => (
              <span key={tag} className="skill-tag">{tag}</span>
            ))}
          </div>
        </div>
      )}

      {activeTab === 'reviews' && (
        <div className="review-list">
          {(worker as any).reviews?.map((r: any) => (
            <div key={r.id} className="review">
              <div className="review-top">
                <strong>{r.reviewer?.full_name}</strong>
                <span>{'★'.repeat(r.rating)}</span>
              </div>
              <p>{r.comment}</p>
            </div>
          ))}
        </div>
      )}

      {/* ── CONTACT MODAL ── */}
      {showContact && (
        <ContactModal
          workerName={worker.full_name}
          onPay={(phone) => { setShowContact(false); handleContact(phone) }}
          onClose={() => setShowContact(false)}
        />
      )}
    </div>
  )
}

// ── HELPERS ───────────────────────────────────────────
function initials(name: string) {
  return name.split(' ').map(n => n[0]).join('').slice(0, 2).toUpperCase()
}

function avatarColor(name: string) {
  const colors = [
    '#1A7A4A,#145E38', '#2563EB,#7C3AED', '#B45309,#D97706',
    '#DC2626,#C4541A', '#0891B2,#059669', '#7C3AED,#EC4899'
  ]
  const idx = name.charCodeAt(0) % colors.length
  return colors[idx]
}

      `)
      .in('job_id',
        (await getEmployerJobs(profile.id)).jobs.map(j => j.id)
      )
      .order('created_at', { ascending: false }),

    supabase.from('wallets')
      .select('balance')
      .eq('user_id', profile.id)
      .single(),
  ])

  return (
    <EmployerDashboard
      profile={profile}
      jobs={jobsResult.jobs}
      applications={applicationsResult.data ?? []}
      walletBalance={walletResult.data?.balance ?? 0}
    />
  )
}

ISR Revalidation

Worker profile pages are statically generated for the top 100 workers. New workers or profile updates trigger revalidation via the API.

app/api/shinga/revalidate/route.ts
import { NextRequest, NextResponse } from 'next/server'
import { revalidatePath }            from 'next/cache'

// Called from Supabase webhooks when a worker profile is updated
export async function POST(req: NextRequest) {
  const secret = req.headers.get('x-revalidate-secret')
  if (secret !== process.env.REVALIDATE_SECRET)
    return NextResponse.json({ error: 'Invalid secret' }, { status: 401 })

  const { workerId, type } = await req.json()

  if (type === 'worker' && workerId) {
    revalidatePath(`/workers/${workerId}`)
  }
  if (type === 'jobs') {
    revalidatePath('/')
    revalidatePath('/jobs')
  }

  return NextResponse.json({ revalidated: true })
}

Security & RLS Summary

✅ What Supabase RLS enforces automatically
Employers can only read/update their own jobs and see applications to their jobs.
Workers can only read/update their own profile. Their phone number is never returned in any public query — only through the /api/shinga/workers/[id]/contact endpoint after a valid payment reference is verified.
Applications are only visible to the job's employer or the applying worker.
Payments are only visible to the user who made them.
Wallets are only readable by their owner.
Admin routes require role = 'admin' enforced in middleware.
⚠️ Phone number protection — critical
The worker's phone column must be excluded from all public queries. Only the /api/shinga/workers/[id]/contact route returns it — and only after verifying the payment reference is completed and belongs to the requesting user. This is enforced both in RLS and in the API route logic.
app/api/shinga/workers/[id]/contact/route.ts — phone reveal
import { NextRequest, NextResponse } from 'next/server'
import { getUser }            from '@/lib/shinga/auth'
import { createServerClient } from '@/lib/shinga/supabase'

export async function GET(
  req: NextRequest,
  { params }: { params: { id: string } }
) {
  const user = await getUser()
  if (!user) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })

  const ref      = req.nextUrl.searchParams.get('ref')
  const supabase = await createServerClient()

  // 1. Verify the payment exists, is completed, belongs to this user,
  //    and was for this specific worker
  const { data: payment } = await supabase
    .from('payments')
    .select('status, user_id, metadata')
    .eq('reference_id', ref)
    .single()

  if (!payment || payment.status !== 'completed')
    return NextResponse.json({ error: 'Payment not verified' }, { status: 402 })
  if (payment.user_id !== user.id)
    return NextResponse.json({ error: 'Forbidden' }, { status: 403 })
  if (payment.metadata?.worker_id !== params.id)
    return NextResponse.json({ error: 'Wrong worker' }, { status: 403 })

  // 2. Fetch phone — service role bypasses RLS for this one read
  const { data: worker } = await supabase
    .from('workers')
    .select('phone, full_name')
    .eq('id', params.id)
    .single()

  return NextResponse.json({ phone: worker?.phone })
}
