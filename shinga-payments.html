MoMo Payment Integration

Complete implementation for MTN Mobile Money and Airtel Money in Rwanda. Covers every SHINGA payment trigger — job posting fees, application fees, contact fees, verification, and credit wallet top-ups.

✅ What this covers
MTN MoMo API (Rwanda) — Collection API for pulling payments from user wallets.
Airtel Money Rwanda — Collection API via Flutterwave or direct Airtel API.
Webhook handling — Receiving payment confirmations asynchronously.
Idempotency — Safe retries, no double charges.
Credit wallet system — Prepaid balance that deducts on every action.

Payment Providers

📱
MTN MoMo Rwanda
Primary
Direct MTN MoMo Collections API. Most common wallet in Rwanda. Apply at momodeveloper.mtn.com. Test sandbox available instantly.
Requires business registration in Rwanda. Processing time: 1–3 days for production keys.
📡
Airtel Money Rwanda
Secondary
Via Flutterwave (recommended) or direct Airtel API. Flutterwave handles both MTN and Airtel under one integration.
Flutterwave alternative: one API key for both networks. Slightly higher fees (~1.4%) but faster setup.
💡 Recommended approach for launch
Start with Flutterwave — one integration handles MTN MoMo + Airtel Money + bank cards. Once you hit volume, add direct MTN MoMo API to reduce fees. This guide implements both but shows Flutterwave as the primary path.

Payment Flow

SHINGA uses a push collection model — the user approves a debit from their own MoMo wallet. No card details needed.

1
User clicks "Pay & Post" / "Pay & Apply"
Frontend calls your Next.js API route with the action type and amount.
POST /api/payments/initiate
2
Server creates a payment record
Generates a unique reference_id, saves to Supabase payments table with status pending.
DB: INSERT payments
3
Server calls Flutterwave / MTN API
Sends the user's phone number and amount. MoMo sends a USSD push to their phone.
POST flutterwave.com/v3/charges
4
User approves on their phone
User enters MoMo PIN on their handset. Takes 10–60 seconds.
User action
5
Webhook fires to your server
Flutterwave/MTN sends a POST to /api/payments/webhook with payment status.
POST /api/payments/webhook
6
Server verifies and fulfills
Verifies webhook signature. Updates payment status to completed. Triggers the action (job goes live, application sent, contact unlocked).
DB: UPDATE payments + fulfill action
7
Frontend polls for confirmation
Frontend polls /api/payments/status/:ref every 3s until confirmed or timed out.
GET /api/payments/status/:ref

Core Payment Library

Create lib/shinga/payments.ts — the central file for all payment operations.

lib/shinga/payments.ts
// ═══════════════════════════════════════════════════
// SHINGA — Payment Library (Flutterwave + MTN MoMo)
// ═══════════════════════════════════════════════════

import { createServerClient } from './supabase'
import { v4 as uuidv4 }    from 'uuid'

export type PaymentAction =
  | 'post_job'
  | 'apply_job'
  | 'contact_worker'
  | 'verification'
  | 'wallet_topup'
  | 'contact_bundle_5'
  | 'contact_bundle_15'

export interface InitiatePaymentInput {
  userId:       string
  phone:        string       // +250XXXXXXXXX
  amount:       number       // in RWF
  action:       PaymentAction
  metadata:     Record<string, unknown>  // job_id, worker_id, etc.
  provider?:    'flutterwave' | 'mtn'
}

export interface PaymentRecord {
  id:           string
  reference_id: string
  user_id:      string
  action:       PaymentAction
  amount:       number
  currency:     string
  phone:        string
  status:       'pending' | 'completed' | 'failed' | 'cancelled'
  provider:     string
  metadata:     Record<string, unknown>
  fulfilled_at: string | null
  created_at:   string
}

// ── INITIATE PAYMENT ──────────────────────────────────

export async function initiatePayment(
  input: InitiatePaymentInput
): Promise<{ referenceId: string; error: string | null }> {
  const supabase   = await createServerClient()
  const referenceId = uuidv4()
  const provider    = input.provider ?? 'flutterwave'

  // 1. Save payment record (status: pending)
  const { error: dbErr } = await supabase
    .from('payments')
    .insert({
      reference_id: referenceId,
      user_id:      input.userId,
      action:       input.action,
      amount:       input.amount,
      currency:     'RWF',
      phone:        input.phone,
      status:       'pending',
      provider,
      metadata:     input.metadata,
    })

  if (dbErr) return { referenceId: '', error: dbErr.message }

  // 2. Call payment provider
  const result = provider === 'mtn'
    ? await callMtnMomo(referenceId, input)
    : await callFlutterwave(referenceId, input)

  if (result.error) {
    // Mark as failed immediately if API call itself fails
    await supabase.from('payments').update({ status: 'failed' })
      .eq('reference_id', referenceId)
    return { referenceId: '', error: result.error }
  }

  return { referenceId, error: null }
}

// ── FLUTTERWAVE ────────────────────────────────────────

async function callFlutterwave(
  referenceId: string,
  input: InitiatePaymentInput
): Promise<{ error: string | null }> {
  const res = await fetch('https://api.flutterwave.com/v3/charges?type=mobile_money_rwanda', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${process.env.FLW_SECRET_KEY}`,
    },
    body: JSON.stringify({
      tx_ref:   referenceId,
      amount:   input.amount,
      currency: 'RWF',
      email:    `${input.userId}@shinga.rw`,   // required by FLW
      phone_number: input.phone,
      fullname: 'SHINGA User',
      redirect_url: `${process.env.NEXT_PUBLIC_APP_URL}/payment/confirm`,
      meta: { action: input.action, ...input.metadata },
      network: detectNetwork(input.phone),  // MTN or AIRTEL
    }),
  })

  const data = await res.json()
  if (data.status !== 'success') {
    return { error: data.message || 'Flutterwave payment initiation failed' }
  }
  return { error: null }
}

// ── MTN MOMO DIRECT ────────────────────────────────────

async function callMtnMomo(
  referenceId: string,
  input: InitiatePaymentInput
): Promise<{ error: string | null }> {
  const token = await getMtnAccessToken()
  if (!token) return { error: 'MTN auth failed' }

  const res = await fetch(
    `https://sandbox.momodeveloper.mtn.com/collection/v1_0/requesttopay`,
    {
      method: 'POST',
      headers: {
        'Authorization':     `Bearer ${token}`,
        'X-Reference-Id':    referenceId,
        'X-Target-Environment': process.env.MTN_ENVIRONMENT || 'sandbox',
        'Ocp-Apim-Subscription-Key': process.env.MTN_SUBSCRIPTION_KEY!,
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({
        amount:     String(input.amount),
        currency:   'RWF',
        externalId: referenceId,
        payer: { partyIdType: 'MSISDN', partyId: input.phone.replace('+', '') },
        payerMessage: `SHINGA: ${input.action.replace(/_/g, ' ')}`,
        payeeNote:    'SHINGA Rwanda Payment',
      }),
    }
  )

  if (res.status !== 202) {
    const err = await res.text()
    return { error: `MTN error: ${err}` }
  }
  return { error: null }
}

async function getMtnAccessToken(): Promise<string | null> {
  const creds = Buffer.from(`${process.env.MTN_API_USER_ID}:${process.env.MTN_API_KEY}`).toString('base64')
  const res = await fetch(
    'https://sandbox.momodeveloper.mtn.com/collection/token/',
    { method: 'POST', headers: { 'Authorization': `Basic ${creds}`, 'Ocp-Apim-Subscription-Key': process.env.MTN_SUBSCRIPTION_KEY! } }
  )
  const data = await res.json()
  return data.access_token || null
}

// ── WEBHOOK VERIFICATION ───────────────────────────────

export function verifyFlutterwaveWebhook(
  payload: string,
  signature: string
): boolean {
  const crypto   = require('crypto')
  const hash     = crypto
    .createHmac('sha256', process.env.FLW_WEBHOOK_HASH!)
    .update(payload)
    .digest('hex')
  return hash === signature
}

// ── FULFILL ACTION ─────────────────────────────────────
// Called after webhook confirms payment. Performs the
// actual platform action the user paid for.

export async function fulfillPayment(
  referenceId: string
): Promise<{ error: string | null }> {
  const supabase = await createServerClient()

  const { data: payment, error } = await supabase
    .from('payments')
    .select('*')
    .eq('reference_id', referenceId)
    .single()

  if (error || !payment) return { error: 'Payment not found' }
  if (payment.fulfilled_at) return { error: null }  // idempotent

  switch (payment.action as PaymentAction) {
    case 'post_job':
      await supabase.from('jobs')
        .update({ status: 'active' })
        .eq('id', payment.metadata.job_id)
      break

    case 'apply_job':
      await supabase.from('applications').insert({
        job_id:    payment.metadata.job_id,
        worker_id: payment.metadata.worker_id,
        status:    'pending',
      })
      break

    case 'contact_worker':
    case 'contact_bundle_5':
    case 'contact_bundle_15': {
      const qty = payment.action === 'contact_bundle_15' ? 15
                : payment.action === 'contact_bundle_5'  ? 5  : 1
      await supabase.from('contact_credits').insert({
        user_id:  payment.user_id,
        quantity: qty,
      })
      break
    }

    case 'verification':
      await supabase.from('workers').update({
        verified:       true,
        featured_until: new Date(Date.now() + 365 * 24 * 60 * 60 * 1000).toISOString(),
      }).eq('profile_id', payment.user_id)
      break

    case 'wallet_topup':
      await supabase.rpc('increment_wallet', {
        p_user_id: payment.user_id,
        p_amount:  payment.metadata.credit_amount,  // face value, not price paid
      })
      break
  }

  // Mark fulfilled
  await supabase.from('payments').update({
    status:       'completed',
    fulfilled_at: new Date().toISOString(),
  }).eq('reference_id', referenceId)

  return { error: null }
}

// ── HELPERS ────────────────────────────────────────────

function detectNetwork(phone: string): 'MTN' | 'AIRTEL' {
  // Rwanda: MTN = 078/079, Airtel = 072/073
  const n = phone.replace(/\D/g, '').slice(-9)
  return ['78', '79'].some(p => n.startsWith(p)) ? 'MTN' : 'AIRTEL'
}

export async function getPaymentStatus(referenceId: string) {
  const supabase = await createServerClient()
  const { data } = await supabase
    .from('payments')
    .select('status, fulfilled_at, action, amount')
    .eq('reference_id', referenceId)
    .single()
  return data
}

API Routes

Initiate Payment — app/api/payments/initiate/route.ts

route.ts
import { NextRequest, NextResponse } from 'next/server'
import { initiatePayment } from '@/lib/shinga/payments'
import { getUser }         from '@/lib/shinga/auth'
import { PRICES }          from '@/lib/shinga/prices'  // fetched from Supabase

export async function POST(req: NextRequest) {
  const user = await getUser()
  if (!user) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })

  const { action, phone, metadata } = await req.json()

  // Resolve correct amount from live pricing config
  const amountMap: Record<string, number> = {
    post_job_daily:      PRICES.job_post_daily,
    post_job_weekly:     PRICES.job_post_weekly,
    post_job_monthly:    PRICES.job_post_monthly,
    post_job_bulk:       PRICES.job_post_bulk,
    post_job_contract:   PRICES.job_post_contract,
    post_job_permanent:  PRICES.job_post_permanent,
    post_job_task:       PRICES.job_post_task,
    apply_job:           PRICES.application_fee,
    contact_worker:      PRICES.contact_fee_single,
    contact_bundle_5:    PRICES.contact_fee_5,
    contact_bundle_15:   PRICES.contact_fee_15,
    verification:        PRICES.verification_fee,
    wallet_topup:        metadata?.price ?? 0,    // buyer pays price, gets face value
  }

  const amount = amountMap[action]
  if (!amount) return NextResponse.json({ error: 'Invalid action' }, { status: 400 })

  const { referenceId, error } = await initiatePayment({
    userId: user.id, phone, amount,
    action, metadata: metadata ?? {},
  })

  if (error) return NextResponse.json({ error }, { status: 400 })

  return NextResponse.json({ referenceId, amount })
}

Webhook Handler — app/api/payments/webhook/route.ts

webhook/route.ts
import { NextRequest, NextResponse } from 'next/server'
import { verifyFlutterwaveWebhook, fulfillPayment } from '@/lib/shinga/payments'
import { sendSms } from '@/lib/shinga/sms'

export async function POST(req: NextRequest) {
  const body      = await req.text()
  const signature = req.headers.get('verif-hash') ?? ''

  // Always verify before doing anything
  if (!verifyFlutterwaveWebhook(body, signature)) {
    return NextResponse.json({ error: 'Invalid signature' }, { status: 401 })
  }

  const event = JSON.parse(body)

  if (event.status === 'successful') {
    const referenceId = event.data.tx_ref

    // Fulfill the platform action
    const { error } = await fulfillPayment(referenceId)
    if (error) return NextResponse.json({ error }, { status: 500 })

    // Send SMS confirmation (Africa's Talking)
    await sendSms(event.data.customer.phone_number,
      `SHINGA: Payment of RWF ${event.data.amount.toLocaleString()} confirmed. Ref: ${referenceId.slice(0,8)}.`)
  }

  // Always return 200 so Flutterwave stops retrying
  return NextResponse.json({ received: true })
}

// Disable body parsing — we need raw text for signature verification
export const config = { api: { bodyParser: false } }

  _: NextRequest,

Frontend Payment Hook

Drop this in hooks/shinga/usePayment.ts. It handles initiation, polling, timeout, and success/failure callbacks.

hooks/shinga/usePayment.ts
'use client'

import { useState, useCallback, useRef } from 'react'

type PaymentStatus = 'idle' | 'pending' | 'completed' | 'failed' | 'timeout'

interface UsePaymentOptions {
  onSuccess?: (referenceId: string) => void
  onFailure?: (error: string) => void
  timeoutMs?: number   // default 120 000 ms (2 min)
  pollMs?:    number   // default 3 000 ms
}

export function usePayment(options: UsePaymentOptions = {}) {
  const { onSuccess, onFailure, timeoutMs = 120_000, pollMs = 3_000 } = options
  const [status,      setStatus]      = useState<PaymentStatus>('idle')
  const [referenceId, setReferenceId] = useState<string | null>(null)
  const [amount,      setAmount]      = useState<number>(0)
  const [error,       setError]       = useState<string | null>(null)
  const pollRef = useRef<ReturnType<typeof setInterval> | null>(null)
  const timerRef= useRef<ReturnType<typeof setTimeout> | null>(null)

  const stopPolling = useCallback(() => {
    if (pollRef.current)  clearInterval(pollRef.current)
    if (timerRef.current) clearTimeout(timerRef.current)
  }, [])

  const pay = useCallback(async (input: {
    action:   string
    phone:    string
    metadata?: Record<string, unknown>
  }) => {
    setStatus('pending')
    setError(null)

    // 1. Initiate payment
    const res  = await fetch('/api/payments/initiate', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(input),
    })
    const data = await res.json()
    if (data.error) { setStatus('failed'); setError(data.error); onFailure?.(data.error); return }

    setReferenceId(data.referenceId)
    setAmount(data.amount)

    // 2. Timeout guard
    timerRef.current = setTimeout(() => {
      stopPolling()
      setStatus('timeout')
      onFailure?.('Payment timed out. Check your MoMo balance and try again.')
    }, timeoutMs)

    // 3. Poll for confirmation
    pollRef.current = setInterval(async () => {
      const r    = await fetch(`/api/payments/status/${data.referenceId}`)
      const stat = await r.json()

      if (stat.status === 'completed') {
        stopPolling()
        setStatus('completed')
        onSuccess?.(data.referenceId)
      } else if (stat.status === 'failed') {
        stopPolling()
        setStatus('failed')
        onFailure?.('Payment was declined. Please try again.')
      }
    }, pollMs)
  }, [onSuccess, onFailure, stopPolling, timeoutMs, pollMs])

  return { pay, status, referenceId, amount, error, stopPolling }
}

Usage in a Component

PostJobModal.tsx
'use client'
import { usePayment } from '@/hooks/shinga/usePayment'
import { useAuth }    from '@/hooks/shinga'

export function PostJobModal({ jobId }: { jobId: string }) {
  const { profile } = useAuth()
  const { pay, status, amount } = usePayment({
    onSuccess: (ref) => {
      toast('✓ Payment confirmed! Your job is now live.')
      router.push(`/dashboard/my-jobs`)
    },
    onFailure: (err) => toast(err, 'error'),
  })

  const handlePay = () => pay({
    action:   'post_job_daily',
    phone:    profile?.phone ?? '',
    metadata: { job_id: jobId },
  })

  if (status === 'pending') return (
    <div className="payment-pending">
      <div className="spinner" />
      <p>Check your phone — approve the MoMo payment of RWF {amount.toLocaleString()}</p>
      <small>This will disappear automatically once confirmed</small>
    </div>
  )

  return <button onClick={handlePay}>Pay & Post Job</button>
}

Database Schema Addition

Add to shinga-schema.sql — run in Supabase SQL Editor.

SQL — payments + wallet tables
-- ── PAYMENTS ─────────────────────────────────────────
create table payments (
  id            uuid primary key default uuid_generate_v4(),
  reference_id  text unique not null,
  user_id       uuid references profiles(id) on delete set null,
  action        text not null,
  amount        numeric(12,2) not null,
  currency      text default 'RWF',
  phone         text,
  provider      text default 'flutterwave',
  status        text default 'pending'
                  check (status in ('pending','completed','failed','cancelled')),
  metadata      jsonb default '{}',
  fulfilled_at  timestamptz,
  created_at    timestamptz default now()
);

create index payments_user_idx   on payments(user_id);
create index payments_status_idx on payments(status);
create index payments_ref_idx    on payments(reference_id);

-- ── WALLET ───────────────────────────────────────────
create table wallets (
  user_id   uuid primary key references profiles(id) on delete cascade,
  balance   numeric(14,2) default 0 check (balance >= 0),
  updated_at timestamptz default now()
);

-- ── CONTACT CREDITS ──────────────────────────────────
create table contact_credits (
  id         uuid primary key default uuid_generate_v4(),
  user_id    uuid references profiles(id) on delete cascade,
  quantity   integer not null,
  used       integer default 0,
  created_at timestamptz default now()
);

-- ── WALLET INCREMENT FUNCTION ─────────────────────────
create or replace function increment_wallet(
  p_user_id uuid, p_amount numeric
) returns void as $$
begin
  insert into wallets (user_id, balance)
  values (p_user_id, p_amount)
  on conflict (user_id) do update
    set balance    = wallets.balance + p_amount,
        updated_at = now();
end;
$$ language plpgsql;

-- ── RLS ───────────────────────────────────────────────
alter table payments        enable row level security;
alter table wallets         enable row level security;
alter table contact_credits enable row level security;

create policy "own_payments" on payments
  for select using (user_id = auth.uid());

create policy "own_wallet" on wallets
  for all using (user_id = auth.uid());

create policy "own_credits" on contact_credits
  for select using (user_id = auth.uid());
